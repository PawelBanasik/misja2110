    // --- 1. GWIEZDNE TŁO (Canvas Starfield) ---
    const canvas = document.getElementById('starfield');
    const ctx = canvas.getContext('2d');
    let stars = [];
    let shootingStars = [];
    let width, height;

    function resizeCanvas() {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
      initStars();
    }

    function initStars() {
      stars = [];
      const numStars = Math.floor((width * height) / 2500);
      for (let i = 0; i < numStars; i++) {
        stars.push({
          x: Math.random() * width,
          y: Math.random() * height,
          size: Math.random() * 1.8 + 0.3,
          alpha: Math.random() * 0.8 + 0.2,
          speed: Math.random() * 0.02 + 0.005,
          color: Math.random() > 0.3 ? '#ffffff' : (Math.random() > 0.5 ? '#a5b4fc' : '#f472b6')
        });
      }
    }

    function createShootingStar() {
      if (shootingStars.length < 2 && Math.random() < 0.015) {
        shootingStars.push({
          x: Math.random() * width,
          y: Math.random() * (height / 2),
          length: Math.random() * 80 + 40,
          speed: Math.random() * 8 + 6,
          angle: Math.PI / 4 + (Math.random() * 0.2 - 0.1),
          alpha: 1
        });
      }
    }

    function renderStarfield() {
      ctx.clearRect(0, 0, width, height);

      // Rysowanie gwiazd stałych
      for (let star of stars) {
        star.alpha += star.speed;
        if (star.alpha > 1 || star.alpha < 0.2) star.speed = -star.speed;

        ctx.fillStyle = star.color;
        ctx.globalAlpha = Math.max(0.1, Math.min(1, star.alpha));
        ctx.beginPath();
        ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
        ctx.fill();
      }

      // Rysowanie spadających gwiazd
      createShootingStar();
      for (let i = shootingStars.length - 1; i >= 0; i--) {
        const s = shootingStars[i];
        ctx.save();
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 1.8;
        ctx.globalAlpha = s.alpha;

        const tailX = s.x - Math.cos(s.angle) * s.length;
        const tailY = s.y - Math.sin(s.angle) * s.length;

        const grad = ctx.createLinearGradient(s.x, s.y, tailX, tailY);
        grad.addColorStop(0, 'rgba(255, 255, 255, 1)');
        grad.addColorStop(1, 'rgba(192, 132, 252, 0)');
        ctx.strokeStyle = grad;

        ctx.beginPath();
        ctx.moveTo(s.x, s.y);
        ctx.lineTo(tailX, tailY);
        ctx.stroke();
        ctx.restore();

        s.x += Math.cos(s.angle) * s.speed;
        s.y += Math.sin(s.angle) * s.speed;
        s.alpha -= 0.015;

        if (s.alpha <= 0 || s.x > width || s.y > height) {
          shootingStars.splice(i, 1);
        }
      }

      requestAnimationFrame(renderStarfield);
    }

    let audioCtx = null;
    let isAudioPlaying = false;
    let ambientDroneOsc1 = null;

    function initAudio() {
      if (!audioCtx) {

    function toggleAmbientSound() {
      initAudio();
      const audioBtn = document.getElementById('audioBtn');
      const audioIcon = document.getElementById('audioIcon');
      const audioLabel = document.getElementById('audioLabel');

      if (!isAudioPlaying) {
        // Dźwięk tła: łagodny kosmiczny drone
        ambientGain = audioCtx.createGain();
        ambientGain.gain.setValueAtTime(0.04, audioCtx.currentTime);

        ambientDroneOsc1 = audioCtx.createOscillator();
        ambientDroneOsc1.type = 'sine';
        ambientDroneOsc1.frequency.setValueAtTime(110, audioCtx.currentTime); // A2

        ambientDroneOsc2 = audioCtx.createOscillator();
        ambientDroneOsc2.type = 'sine';
        ambientDroneOsc2.frequency.setValueAtTime(164.81, audioCtx.currentTime); // E3

        ambientDroneOsc1.connect(ambientGain);
        ambientDroneOsc2.connect(ambientGain);
        ambientGain.connect(audioCtx.destination);

        ambientDroneOsc1.start();
        ambientDroneOsc2.start();

        isAudioPlaying = true;
        audioIcon.innerText = '🔊';
        audioLabel.innerText = 'Dźwięk: ON';
        audioBtn.classList.add('border-indigo-400', 'text-white');
      } else {
        if (ambientGain) {
          ambientGain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.8);
          setTimeout(() => {
            if (ambientDroneOsc1) ambientDroneOsc1.stop();
            if (ambientDroneOsc2) ambientDroneOsc2.stop();
          }, 800);
        }
        isAudioPlaying = false;
        audioIcon.innerText = '🔈';
        audioLabel.innerText = 'Dźwięk: OFF';
        audioBtn.classList.remove('border-indigo-400', 'text-white');
      }
    }

    // Dźwięk skanera / kosmicznego gongu
    function playCosmicBeep(freq = 587.33, duration = 0.5) {
      if (!audioCtx) initAudio();
      if (!audioCtx) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + duration);
    }
